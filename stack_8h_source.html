71static inline int lfi(resize)(lfi_self *s, size_t new_cap)
72{
73 if (s->lfi(cap) == new_cap)
74 return 0;
75
76 lfi_key *old_data = s->lfi(data);
77
78 s->lfi(data) = (lfi_key *) realloc(s->lfi(data),
79 new_cap * sizeof(lfi_key));
80
81 if (s->lfi(data) == NULL) {
82 s->lfi(data) = old_data;
83 return 1;
84 }
85
86 s->lfi(cap) = new_cap;
87 return 0;
88}

153lfi_wur static inline lfi_key *lfi_memb(multipush)(lfi_self *s,
154 lfi_key const items[],
155 size_t count)
156{
157 while (s->lfi(len) + count > s->lfi(cap))
158 lfi(resize)(s, s->lfi(cap) * 2);
159
160 if (s->lfi(len) + count > s->lfi(cap))
161 return NULL;
162
163 lfi_key *pushed_bottom = &s->lfi(data)[s->lfi(len)];
164
165 if (items != NULL) {
166 for (size_t i = 0; i < count; i++)
167 s->lfi(data)[s->lfi(len)++] = items[i];
168 } else {
169 s->lfi(len) += count;
170 }
171
172 return pushed_bottom;
173}
