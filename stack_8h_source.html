74static inline int lfi(resize)(lfi_self *s, size_t new_cap)
75{
76 if (s->lfi(cap) == new_cap)
77 return 0;
78
79 lfi_key *old_data = s->lfi(data);
80
81 s->lfi(data) = (lfi_key *) realloc(s->lfi(data),
82 new_cap * sizeof(lfi_key));
83
84 if (s->lfi(data) == NULL) {
85 s->lfi(data) = old_data;
86 return 1;
87 }
88
89 s->lfi(cap) = new_cap;
90 return 0;
91}

156lfi_wur static inline lfi_key *lfi_memb(multipush)(lfi_self *s,
157 lfi_key const items[],
158 size_t count)
159{
160 while (s->lfi(len) + count > s->lfi(cap))
161 lfi(resize)(s, s->lfi(cap) * 2);
162
163 if (s->lfi(len) + count > s->lfi(cap))
164 return NULL;
165
166 lfi_key *pushed_bottom = &s->lfi(data)[s->lfi(len)];
167
168 if (items != NULL) {
169 for (size_t i = 0; i < count; i++)
170 s->lfi(data)[s->lfi(len)++] = items[i];
171 } else {
172 s->lfi(len) += count;
173 }
174
175 return pushed_bottom;
176}
