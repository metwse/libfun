115static uint64_t lfi(hash)(const lfi_key *key, size_t key_len)
116{
117 const char *mapped_key_bytes;
118 size_t mapped_key_len;
119 uint64_t hash;
120#ifdef fmap_key
121 fmap_key(key, key_len, &mapped_key_bytes, &mapped_key_len);
122#else
123 mapped_key_bytes = (const char *) key;
124 mapped_key_len = key_len;
125#endif
126#ifdef fuse_hash
127 hash = fuse_hash(mapped_key_bytes, mapped_key_len);
128#else
129 hash = lfi_g(fnv_hash)(mapped_key_bytes, mapped_key_len);
130#endif
131 return hash;
132}

134static inline struct lfi(entry) **lfi(get_slot)(lfi_self *m,
135 const lfi_key *key,
136 size_t key_len)
137{
138 uint64_t hash = lfi(hash)(key, key_len);
139
140 size_t start_i = hash % m->lfi(cap);
141 size_t i = start_i;
142
143 do {
144 struct lfi(entry) **slot = &m->lfi(entries)[i];
145
146 if (*slot == NULL)
147 return NULL;
148 else if (*slot != LFI_HASHMAP_TOMBSTONE &&
149 (*slot)->key_len == key_len &&
150 memcmp((*slot)->key, key, key_len) == 0)
151 return slot;
152 else
153 i++;
154
155 i %= m->lfi(cap);
156 } while (i != start_i);
157
158 /* Hashmap filled up with tombstones */
159 return NULL;
160}

162static inline void lfi(insert_entry)(lfi_self *m, struct lfi(entry) *e)
163{
164 lfi_assert(m->lfi(cap) > m->lfi(used),
165 "call to insert_entry on full map");
166
167 uint64_t hash = lfi(hash)(e->key, e->key_len);
168 size_t start_i = hash % m->lfi(cap);
169 size_t i = start_i;
170
171 do {
172 struct lfi(entry) **slot = &m->lfi(entries)[i];
173
174 if (*slot == NULL || *slot == LFI_HASHMAP_TOMBSTONE) {
175 m->lfi(used)++;
176 *slot = e;
177
178 return;
179 } else {
180 i++;
181 }
182
183 i %= m->lfi(cap);
184 } while (i != start_i);
185
186 lfi_unreachable; // GCOVR_EXCL_LINE: unreachable
187}

189static inline int lfi(resize)(lfi_self *m, size_t new_cap)
190{
191 if (m->lfi(cap) == new_cap)
192 return 0;
193
194 struct lfi(entry) **old_entries = m->lfi(entries);
195
196 m->lfi(entries) = calloc(new_cap, sizeof(struct lfi(entry) *));
197
198 if (m->lfi(entries) == NULL) {
199 m->lfi(entries) = old_entries;
200 return 1;
201 }
202
203 size_t old_cap = m->lfi(cap);
204 m->lfi(used) = 0;
205 m->lfi(cap) = new_cap;
206
207 for (size_t i = 0; i < old_cap; i++) {
208 struct lfi(entry) *e = old_entries[i];
209
210 if (e != NULL)
211 lfi(insert_entry)(m, e);
212 }
213
214 free(old_entries);
215 return 0;
216}

218static inline struct lfi(entry) *lfi(insert)(lfi_self *m,
219 const lfi_key *key,
220 size_t key_len,
221 lfi_value const *value)
222{
223 if (m->lfi(cap) * 3 < m->lfi(used) * 4) {
224 size_t new_cap = m->lfi(cap) * 2;
225
226 lfi(resize)(m, new_cap);
227 }
228
229 if (m->lfi(cap) == m->lfi(used))
230 return NULL;
231
232 lfi_assert(lfi(get_slot)(m, key, key_len) == NULL,
233 "hmap contains the element");
234
235 struct lfi(entry) *e = (struct lfi(entry) *)
236 malloc(sizeof(struct lfi(entry)) + key_len);
237
238 e->key_len = key_len;
239 memcpy(e->key, key, key_len);
240
241 if (value != NULL)
242 e->value = *value;
243
244 lfi(insert_entry)(m, e);
245
246 return e;
247}
