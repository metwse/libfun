132static uint64_t lfi(hash)(const lfi_key *key, size_t key_len)
133{
134 const char *mapped_key_bytes;
135 size_t mapped_key_len;
136 uint64_t hash;
137#ifdef fmap_key
138 fmap_key(key, key_len, &mapped_key_bytes, &mapped_key_len);
139#else
140 mapped_key_bytes = (const char *) key;
141 mapped_key_len = key_len;
142#endif
143#ifdef fuse_hash
144 hash = fuse_hash(mapped_key_bytes, mapped_key_len);
145#else
146 hash = lfi_g(fnv_hash)(mapped_key_bytes, mapped_key_len);
147#endif
148 return hash;
149}

151static inline struct lfi(entry) **lfi(get_entry)(lfi_self *m,
152 const lfi_key *key,
153 size_t key_len)
154{
155 uint64_t hash = lfi(hash)(key, key_len);
156
157 size_t start_i = hash % m->lfi(cap);
158 size_t i = start_i;
159
160 do {
161 struct lfi(entry) **slot = &m->lfi(entries)[i];
162
163 if (*slot == NULL)
164 return NULL;
165 else if (*slot != LFI_HASHMAP_TOMBSTONE &&
166 (*slot)->key_len == key_len &&
167 memcmp((*slot)->key, key, key_len) == 0)
168 return slot;
169 else
170 i++;
171
172 i %= m->lfi(cap);
173 } while (i != start_i);
174
175 /* Hashmap filled up with tombstones */
176 return NULL;
177}

179static inline void lfi(insert_entry)(lfi_self *m, struct lfi(entry) *e)
180{
181 lfi_assert(m->lfi(cap) > m->lfi(used),
182 "call to insert_entry on full map");
183
184 uint64_t hash = lfi(hash)(e->key, e->key_len);
185 size_t start_i = hash % m->lfi(cap);
186 size_t i = start_i;
187
188 do {
189 struct lfi(entry) **slot = &m->lfi(entries)[i];
190
191 if (*slot == NULL || *slot == LFI_HASHMAP_TOMBSTONE) {
192 m->lfi(used)++;
193 *slot = e;
194
195 return;
196 } else {
197 i++;
198 }
199
200 i %= m->lfi(cap);
201 } while (i != start_i);
202
203 lfi_unreachable; // GCOVR_EXCL_LINE: unreachable
204}

206static inline int lfi(resize)(lfi_self *m, size_t new_cap)
207{
208 if (m->lfi(cap) == new_cap)
209 return 0;
210
211 struct lfi(entry) **old_entries = m->lfi(entries);
212
213 m->lfi(entries) = calloc(new_cap, sizeof(struct lfi(entry) *));
214
215 if (m->lfi(entries) == NULL) {
216 m->lfi(entries) = old_entries;
217 return 1;
218 }
219
220 size_t old_cap = m->lfi(cap);
221 m->lfi(used) = 0;
222 m->lfi(cap) = new_cap;
223
224 for (size_t i = 0; i < old_cap; i++) {
225 struct lfi(entry) *e = old_entries[i];
226
227 if (e != NULL)
228 lfi(insert_entry)(m, e);
229 }
230
231 free(old_entries);
232 return 0;
233}

235static inline struct lfi(entry) *lfi(insert)(lfi_self *m,
236 const lfi_key *key,
237 size_t key_len,
238 lfi_value const *value)
239{
240 if (m->lfi(cap) * 3 < m->lfi(used) * 4) {
241 size_t new_cap = m->lfi(cap) * 2;
242
243 lfi(resize)(m, new_cap);
244 }
245
246 if (m->lfi(cap) == m->lfi(used))
247 return NULL;
248
249 lfi_assert(lfi(get_entry)(m, key, key_len) == NULL,
250 "hmap contains the element");
251
252 struct lfi(entry) *e = (struct lfi(entry) *)
253 malloc(sizeof(struct lfi(entry)) + key_len);
254
255 e->key_len = key_len;
256 memcpy(e->key, key, key_len);
257
258 if (value != NULL)
259 e->value = *value;
260
261 lfi(insert_entry)(m, e);
262
263 return e;
264}
